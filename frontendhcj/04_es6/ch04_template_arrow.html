템플릿 리터럴 — 백틱 문자열

const port = 8080;
const domain = 'mydomain';

const url = 'http://' + domain + ':' + port;      // 예전 방식
const url_after = `http://${domain}:${port}`;     // 템플릿 리터럴
알아두기 백틱(`)은 키보드 숫자 1 왼쪽, Tab 위에 있는 키입니다. 작은따옴표(')와 혼동하면 ${ } 가 변수로 해석되지 않고 글자 그대로 출력됩니다.

화살표 함수 — 축약 단계

function add(n1, n2) { return n1 + n2; }   // (1) 전통적인 선언
const add2 = (n1, n2) => n1 + n2;         // (2) 중괄호와 return 생략
const add3 = (n1, n2) => (n1 + n2);       // (3) 괄호로 감싸도 같다
const add4 = (n1, n2) => {                // (4) 중괄호를 쓰면 return 필요
    let temp = n1 + 10;
    return temp + n2;
};
규칙설명
매개변수가 하나면괄호를 생략할 수 있다 : n => n * 2
매개변수가 없으면빈 괄호가 필요하다 : () => "지금"
몸통이 한 줄이면중괄호와 return 을 동시에 생략할 수 있다
중괄호를 쓰면return 을 반드시 직접 써야 한다
객체를 돌려주려면괄호로 감싼다 : () => ({ ok: true })
자주 나는 오류 () => { ok: true } 는 중괄호를 함수 몸통으로 해석해서 undefined 를 돌려줍니다. 객체를 돌려주려면 괄호로 감싸세요.

위 내용은 페이지를 열자마자 자동으로 실행되어 아래 결과 영역에 나옵니다.

this — 두 함수의 근본적인 차이

const timer = {
    name: '타이머',

    startWrong: function () {
        setTimeout(function () {
            console.log(this.name);   // timer 가 아니다 — this 가 바뀌어 버림
        }, 100);
    },

    startRight: function () {
        setTimeout(() => {
            console.log(this.name);   // '타이머' — 바깥의 this 를 그대로 사용
        }, 100);
    }
};

알아두기 function 으로 만든 함수는 "누가 불렀는가"에 따라 this 가 그때그때 정해지지만, 화살표 함수는 자기 자신의 this 를 아예 갖지 않고 바깥 코드의 this 를 그대로 씁니다. 그래서 콜백에는 거의 언제나 화살표 함수를 씁니다.
미니 과제 js/ch04_template_arrow.js 에서 startRight 의 화살표 함수를 function () { ... } 로 바꿔 보세요. 두 출력이 똑같아지는 것을 확인하면 화살표 함수가 무엇을 해결했는지 알 수 있습니다.
