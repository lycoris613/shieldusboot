8장. 모듈 — export 와 import

8장 · 04_ECMAScript_기초.docx

주의 이 페이지도 반드시 Live Server 로 열어야 합니다. 모듈은 file:/// 주소에서 CORS 정책에 막혀 불러올 수 없습니다.

내보내는 쪽 — js/mortgage.js

// named export : 여러 개 가능
export const calculateMonthlyPayment = (principal, rate, years) => { ... };

// export 를 붙이지 않으면 이 파일 안에서만 쓰입니다
const formatWon = (value) => value.toLocaleString('ko-KR') + '원';

const calculateAmortization = (principal, rate, years) => { ... };

// default export : 파일당 하나
export default calculateAmortization;

가져오는 쪽 — js/ch08_module.js

import calcAmortization from './mortgage.js';                 // default (중괄호 없음)
import { calculateMonthlyPayment } from './mortgage.js';      // named (중괄호)
import * as mortgage from './mortgage.js';                    // 통째로

// 이름이 겹치면 as 로 바꿔서 받는다
import { calculateMonthlyPayment as calcMonthly } from './mortgage.js';
구분내보내기가져오기개수
named exportexport const a = ... import { a } from ...여러 개
default exportexport default a import 아무이름 from ...파일당 하나
알아두기 이 페이지는 <script type="module" src="js/ch08_module.js"> 로 불러옵니다. type="module" 이 없으면 Cannot use import statement outside a module 오류가 납니다. 브라우저에서 직접 쓸 때는 경로에 ./ 를 붙이고 확장자 .js 까지 적어야 합니다.
