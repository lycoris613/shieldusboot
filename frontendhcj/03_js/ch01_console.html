세 가지 방법

<!-- ① 태그에 직접 (권장하지 않음) -->
<button onclick="alert('안녕')">클릭</button>

<!-- ② 내부 : body 끝에 script 태그 -->
<script>
    console.log("안녕");
</script>

<!-- ③ 외부 파일 (실무 표준) -->
<script src="js/ch01_console.js"></script>
알아두기 script 태그는 </body> 바로 앞에 둡니다. 위쪽에 두면 아직 만들어지지도 않은 요소를 찾으려다 null 을 받게 됩니다.
자주 나는 오류 <script src="..." /> 처럼 혼자 닫으면 동작하지 않습니다. 반드시 </script> 로 닫으세요.

console — 가장 중요한 도구

console.log("학생 목록 로드 중...");   // 일반 출력
console.log(student);                   // 객체를 펼쳐서
console.error("Error:", error);         // 빨간색으로 강조
console.table(students);                // 배열을 표로

확인하기 F12 → Console 탭을 열고 위 버튼들을 눌러 보세요. console.table 은 Console 탭에서만 표로 보입니다.
