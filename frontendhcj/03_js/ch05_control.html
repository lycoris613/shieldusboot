if — 실습 3-9 의 모드 분기

if (editingStudentId) {
    updateStudent(editingStudentId, studentData);   // 수정 모드
} else {
    createStudent(studentData);                     // 등록 모드
}
알아두기 변수 하나가 화면 전체의 모드를 결정합니다. 이것이 React 등에서 말하는 "상태(state)"의 가장 원시적인 형태입니다.

early return — 검증 코드의 기본 형태

function validateStudent(student) {
    if (!student.name) {
        alert("이름을 입력해주세요.");
        return false;          // 여기서 함수가 끝남
    }
    return true;               // 모든 검사를 통과
}
알아두기 "문제가 있으면 바로 나가고, 끝까지 살아남으면 통과"로 쓰면 if 를 여러 겹 중첩하지 않아도 되어 읽기 쉽습니다.
