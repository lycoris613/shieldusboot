객체가 들어 있는 배열 — 서버가 돌려주는 형태

const students = [
    { id: 1, name: "홍길동" },
    { id: 2, name: "김코딩" }
];

students.length      // 2
students[0]          // { id: 1, name: "홍길동" }   — 0부터 셉니다
students[0].name     // "홍길동"
